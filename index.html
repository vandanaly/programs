IoT Lab Programs 1–10

Arduino programs, subprograms and quick connection notes based on your lab manual.

Before uploading: Choose the correct board and port in Arduino IDE. For Programs 8 and 9, install the required LCD/DHT libraries. Pin assignments and sensor thresholds follow the manual where available; check your exact module. Relay modules may be active-LOW, so reverse the relay ON/OFF logic if needed. Never connect a pump or mains lamp directly to an Arduino pin; use a correctly rated relay/driver and safe isolation.

1(A) — Two LEDs blinking alternately

Connections: LEDs: D6 and D7, each through a 220Ω resistor to GND.

View Arduino code
Arduino C++
void setup() {
  pinMode(6, OUTPUT);
  pinMode(7, OUTPUT);
}

void loop() {
  digitalWrite(6, HIGH);
  digitalWrite(7, LOW);
  delay(500);  // ON time / duty cycle

  digitalWrite(6, LOW);
  digitalWrite(7, HIGH);
  delay(500);
}

1(B) — LED ON/OFF using microswitch + Serial Monitor

Connections: LED: D7 through 220Ω to GND. Microswitch: D5 input; use the pull-down wiring shown in the manual (or adapt INPUT_PULLUP wiring).

View Arduino code
Arduino C++
#define LED_PIN 7
#define MICROSWITCH_PIN 5

void setup() {
  pinMode(LED_PIN, OUTPUT);
  pinMode(MICROSWITCH_PIN, INPUT);
  Serial.begin(9600);
}

void loop() {
  if (digitalRead(MICROSWITCH_PIN) == HIGH) {
    digitalWrite(LED_PIN, HIGH);
    Serial.println("LED ON (1)");
  } else {
    digitalWrite(LED_PIN, LOW);
    Serial.println("LED OFF (0)");
  }
  delay(200);
}
03

1(C) — LED ON/OFF through Serial Monitor only

Connections: LED: D7 through 220Ω to GND. Open Serial Monitor at 9600 baud and send 1 or 2.

View Arduino code
Arduino C++
const int LED_PIN = 7;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
  Serial.println("Enter 1 to turn LED ON, 2 to turn LED OFF");
}

void loop() {
  if (Serial.available() > 0) {
    int state = Serial.parseInt();

    if (state == 1) {
      digitalWrite(LED_PIN, HIGH);
      Serial.println("LED turned ON");
    } else if (state == 2) {
      digitalWrite(LED_PIN, LOW);
      Serial.println("LED turned OFF");
    }
  }
}

2(A) — Automatic LED fading using PWM

Connections: LED anode: D9 through 220Ω resistor; cathode: GND. D9 supports PWM on Arduino Uno.

View Arduino code
Arduino C++
const int ledPin = 9;
int brightness = 0;
int fadeAmount = 5;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  analogWrite(ledPin, brightness);
  brightness += fadeAmount;

  if (brightness <= 0 || brightness >= 125) {
    fadeAmount = -fadeAmount;
  }
  delay(30);
}

2(B) — Potentiometer LED fading + Serial Monitor

Connections: Potentiometer outer pins: 5V and GND; middle pin: A0. LED: D9 through 220Ω to GND.

View Arduino code
Arduino C++
const int potPin = A0;
const int ledPin = 9;

void setup() {
  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int potValue = analogRead(potPin);       // 0–1023
  int brightness = map(potValue, 0, 1023, 0, 255);
  analogWrite(ledPin, brightness);

  Serial.print("Potentiometer: ");
  Serial.print(potValue);
  Serial.print(" | LED brightness: ");
  Serial.println(brightness);
  delay(300);
}

3(C) — Decade UP/DOWN counter (LED binary display)

Connections: Use four LEDs on D2, D3, D4 and D5, each through a 220Ω resistor to GND. The LEDs show a 4-bit binary count.

View Arduino code
Arduino C++
const int pins[] = {2, 3, 4, 5};

void showNumber(int number) {
  for (int bit = 0; bit < 4; bit++) {
    digitalWrite(pins[bit], (number >> bit) & 1);
  }
}

void setup() {
  for (int i = 0; i < 4; i++) {
    pinMode(pins[i], OUTPUT);
  }
}

void loop() {
  // Count up from 0 to 9
  for (int n = 0; n <= 9; n++) {
    showNumber(n);
    delay(1000);
  }

  // Count down from 9 to 0
  for (int n = 9; n >= 0; n--) {
    showNumber(n);
    delay(1000);
  }
}

4 — IR obstacle detector with LED and buzzer

Connections: Manual code reads the sensor on A0, LED on D7 and buzzer on D5. Sensor VCC → 5V, GND → GND, OUT/AO → A0.

View Arduino code
Arduino C++
const int sensorPin = A0;
const int buzzerPin = 5;
const int ledPin = 7;

void setup() {
  Serial.begin(9600);
  pinMode(buzzerPin, OUTPUT);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  int value = analogRead(sensorPin);
  Serial.println(value);

  // Threshold may need adjustment for your sensor/module.
  if (value < 260) {
    digitalWrite(ledPin, HIGH);
    digitalWrite(buzzerPin, HIGH);
    Serial.println("OBSTACLE DETECTED");
  } else {
    digitalWrite(ledPin, LOW);
    digitalWrite(buzzerPin, LOW);
  }
  delay(200);
}

5 — LDR-based lamp control using relay

Connections: LDR one leg → 5V; other leg → A0 and 10kΩ resistor; resistor other leg → GND. Relay module IN → D8, VCC → 5V, GND → GND.

View Arduino code
Arduino C++
const int ldrPin = A0;
const int relayPin = 8;
const int threshold = 400;

void setup() {
  pinMode(relayPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int lightValue = analogRead(ldrPin);
  Serial.println(lightValue);

  // Manual logic: darker/lower light reading activates relay.
  if (lightValue >= threshold) {
    digitalWrite(relayPin, HIGH);
    Serial.println("CLOUDY / EVENING");
  } else {
    digitalWrite(relayPin, LOW);
    Serial.println("ENOUGH LIGHT / DAYTIME");
  }
  delay(1000);
}

6 — Ultrasonic distance measurement and parking alarm

Connections: HC-SR04: VCC → 5V, GND → GND, TRIG → D6, ECHO → D5. Buzzer → D2 and GND. LED (with resistor) can be connected to D7.

View Arduino code
Arduino C++
const int trigPin = 6;
const int echoPin = 5;
const int buzzerPin = 2;
const int ledPin = 7;

void setup() {
  Serial.begin(9600);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 30000);
  int distance = duration * 0.034 / 2;

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (duration > 0 && distance < 50) {
    digitalWrite(ledPin, HIGH);
    digitalWrite(buzzerPin, HIGH);
    delay(max(50, distance * 3));
    digitalWrite(buzzerPin, LOW);
  } else {
    digitalWrite(ledPin, LOW);
    digitalWrite(buzzerPin, LOW);
    delay(200);
  }
}

Connections: MQ-2 sensor: VCC → 5V, GND → GND, AO → A0. LED → D11 and buzzer → D12 (use suitable driver if required).

View Arduino code
Arduino C++
const int smokePin = A0;

}

9 — Temperature and humidity using DHT11 + I²C LCD

Connections: DHT11: VCC → 5V, GND → GND, DATA → D8. LCD I²C: SDA → A4, SCL → A5 (Arduino Uno). Install DHT sensor library and Adafruit Unified Sensor if prompted.

View Arduino code
Arduino C++
#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define DHTPIN 8
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Serial.begin(9600);
  dht.begin();
  lcd.init();
  lcd.backlight();
}

void loop() {
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();

  if (isnan(humidity) || isnan(temperature)) {
    lcd.clear();
    lcd.print("Sensor error");
    delay(1000);
    return;
  }

  Serial.print("Humidity: ");
  Serial.print(humidity);
  Serial.print("%  Temperature: ");
  Serial.print(temperature);
  Serial.println(" C");

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Temp: ");
  lcd.print(temperature, 1);
  lcd.print((char)223);
  lcd.print("C");

  lcd.setCursor(0, 1);
  lcd.print("Humidity: ");
  lcd.print(humidity, 0);
  lcd.print("%");
  delay(1000);
}

10 — Automatic garden watering using soil-moisture sensor

Connections: Soil sensor VCC → 5V, GND → GND, AO → A0. LEDs: D2 (dry), D3 (moist), D4 (wet), each with resistor. Example relay/driver control pin for pump: D8.

View Arduino code
Arduino C++
const int sensorPin = A0;
const int dryLED = 2;
const int moistLED = 3;
const int wetLED = 4;
const int pumpRelay = 8;

void setup() {
  Serial.begin(9600);
  pinMode(sensorPin, INPUT);
  pinMode(dryLED, OUTPUT);
  pinMode(moistLED, OUTPUT);
  pinMode(wetLED, OUTPUT);
  pinMode(pumpRelay, OUTPUT);
}

void loop() {
  int value = analogRead(sensorPin);
  Serial.print("Soil moisture: ");
  Serial.println(value);

  // Thresholds from the lab manual; calibrate for your sensor.
  if (value >= 1000) {
    digitalWrite(dryLED, HIGH);
    digitalWrite(moistLED, LOW);
    digitalWrite(wetLED, LOW);
    digitalWrite(pumpRelay, HIGH); // water when dry
    Serial.println("Soil is dry - watering ON");
  } else if (value >= 630 && value <= 700) {
    digitalWrite(dryLED, LOW);
    digitalWrite(moistLED, HIGH);
    digitalWrite(wetLED, LOW);
    digitalWrite(pumpRelay, LOW);
    Serial.println("Soil is moist");
  } else if (value <= 600) {
    digitalWrite(dryLED, LOW);
    digitalWrite(moistLED, LOW);
    digitalWrite(wetLED, HIGH);
    digitalWrite(pumpRelay, LOW);
    Serial.println("Soil is wet - watering OFF");
  } else {
    // Intermediate reading: avoid unnecessary watering.
    digitalWrite(dryLED, LOW);
    digitalWrite(moistLED, LOW);
    digitalWrite(wetLED, LOW);
    digitalWrite(pumpRelay, LOW);
    Serial.println("Intermediate reading - check/calibrate sensor");
  }
  delay(1000);
}
